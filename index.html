/* ─────────────── 시트 기본 ─────────────── */ function getSheet(ss) { let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); setupSheet(sheet); } if (sheet.getLastRow() === 0) { setupSheet(sheet); } return sheet; } // 날짜 정규화 — 시트가 날짜를 Date 객체로 자동 변환해도 "YYYY-MM-DD" 로 비교 function normDate(v) { if (v instanceof Date) return Utilities.formatDate(v, "Asia/Seoul", "yyyy-MM-dd"); const m = String(v).match(/(\d{4})\D+(\d{1,2})\D+(\d{1,2})/); if (m) return m[1] + "-" + ("0" + m[2]).slice(-2) + "-" + ("0" + m[3]).slice(-2); return String(v).trim(); } // 이름(A열=0) + 날짜(C열=2) 기준으로 마지막 매칭 행 반환 (1-based, 헤더 포함) function findRow(allData, member, date) { let found = -1; const nd = normDate(date); for (let i = 0; i < allData.length; i++) { if (String(allData[i][0]).trim() === String(member).trim() && normDate(allData[i][2]) === nd) { found = i + 2; } } return found; } function buildRow(d, now) { const qa = d.qa || []; } return out; } const row = buildRow(d, now); // ── 수정 또는 신규 — 이름+날짜 기준 덮어쓰기 ── const targetRow = findRow(allData, d.member, d.date); if (targetRow > 0) { sheet.getRange(targetRow, 3).setNumberFormat("@");
